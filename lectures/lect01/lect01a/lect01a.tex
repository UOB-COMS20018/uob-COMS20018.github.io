\begin{frame}[fragile]
\frametitle{Assessment: Mid-term Video (20\%)}

\begin{itemize}
\item \textbf{Format:} individual, up to 15 minutes, submitted as a video
  file on Blackboard.
\item \textbf{Task:} choose one topic $\to$ one question $\to$ 1--2
  papers to discuss in relation to it.
\item \textbf{Scope:} topics drawn from Weeks 2--5 (Machine Learning
  $\to$ Deep Learning $\to$ LLMs $\to$ Philosophy of AI).
\item \textbf{Timing:} topic/question/paper menu will be released by the end of
  Week 3; deadline is mid Week 7.
\item Full topic list, marking criteria, and submission details will be
  posted on Blackboard.
\end{itemize}

\end{frame}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]
\frametitle{Assessment: Lab-Based Work (20\%)}

\begin{itemize}
\item \textbf{Weekly cycle:} notebook released Wednesday $\to$ you submit
  your completed notebook by \textbf{Sunday 23:59} $\to$ answers released
  the following Monday.
\item \textbf{Assessed weeks:} 3, 4, 7, 8, 9, 11 
\item \textbf{Best-5-of-6:} your 1 lowest-scoring lab is automatically
  dropped, so the 20\% is earned from your best 5 labs, each worth 4\% of the unit.
\item Full submission and marking details will be posted on Blackboard.
\end{itemize}

\end{frame}
